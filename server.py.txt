def init_db():
    conn = sqlite3.connect(str(DB_PATH))
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS transactions (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                    document_type TEXT,
                    full_name TEXT,
                    address TEXT,
                    purpose TEXT,
                    amount_paid REAL
                )''')
    conn.commit()
    conn.close()

def log_transaction(doc_type, name, address, purpose, amount):

        conn.close()
        print("Log: Transaction saved.")
    except Exception as e:
        print(f"Log: DB Error -> {e}")

class CoinAcceptor:
    def __init__(self, pin):
        self.pin = pin
        self.total_inserted = 0
        self.sensor = None
        self.enabled = False 
        self.last_pulse_time = 0
        self._setup_gpio()

    def _setup_gpio(self):
        try:
            from gpiozero import Button
            self.sensor = Button(self.pin, pull_up=True, bounce_time=None)
            self.sensor.when_pressed = self._pulse_callback
            print(f"Log: Coin Slot Active on GPIO {self.pin}")
        except ImportError:
            print("Log: gpiozero not found.")
        except Exception as e:
            print(f"Log: GPIO Error: {e}")

    def _pulse_callback(self, device):
        if not self.enabled: return 
        current_time = time.time()
        if (current_time - self.last_pulse_time) < 0.05: return 
        self.last_pulse_time = current_time

        with _lock:
            self.total_inserted += 1
            print(f"💰 Coin Pulse! Total: {self.total_inserted}")

    def start_accepting(self):
        with _lock: self.enabled = True

    def stop_accepting(self):
        with _lock: self.enabled = False

    def reset_and_disable(self):
        with _lock:
            self.total_inserted = 0
            self.enabled = False

    def get_total(self):
        return self.total_inserted

coin_slot = CoinAcceptor(COIN_PIN)

class CameraManager:
    def __init__(self):
        self.camera_type = "NONE"
        self._detect_camera()

    def _detect_camera(self):
        try:
            from picamera2 import Picamera2
            self.camera_type = "PICAMERA2"
            print("Log: Picamera2 detected.")
            return
        except ImportError: pass
        try:
            import cv2
            cap = cv2.VideoCapture(0, cv2.CAP_V4L2)
            if cap.isOpened():
                self.camera_type = "OPENCV"
                cap.release()
                print("Log: USB Camera detected.")
                return
        except ImportError: pass
        self.camera_type = "MOCK"
        print("Log: Using Mock Camera.")

    def run(self):
        global _latest_jpeg
        if self.camera_type == "PICAMERA2":
            try:
                from picamera2 import Picamera2
                import cv2
                import numpy as np
                picam = Picamera2()
                config = picam.create_preview_configuration(main={"size": DEFAULT_PREVIEW_SIZE})
                picam.configure(config)
                picam.start()
                picam.set_controls({"AfMode": 2})
                while not _stop_event.is_set():
                    arr = picam.capture_array()
                    if not arr.flags['C_CONTIGUOUS']: arr = np.ascontiguousarray(arr)
                    _, jpg = cv2.imencode('.jpg', arr, [int(cv2.IMWRITE_JPEG_QUALITY), DEFAULT_JPEG_QUALITY])
                    with _lock: _latest_jpeg = jpg.tobytes()
                    time.sleep(0.04)
            except: self.camera_type = "MOCK"; self.run()
        elif self.camera_type == "OPENCV":
            import cv2
            cap = cv2.VideoCapture(0, cv2.CAP_V4L2)
            cap.set(cv2.CAP_PROP_AUTOFOCUS, 1) 
            while not _stop_event.is_set():
                ret, frame = cap.read()
                if ret:
                    _, jpg = cv2.imencode('.jpg', frame, [int(cv2.IMWRITE_JPEG_QUALITY), DEFAULT_JPEG_QUALITY])
                    with _lock: _latest_jpeg = jpg.tobytes()
                    time.sleep(0.04)
                else: cap.release(); time.sleep(1); cap.open(0, cv2.CAP_V4L2)
        if self.camera_type == "MOCK" or _latest_jpeg is None:
            import cv2
            import numpy as np
            while not _stop_event.is_set():
                img = np.zeros((720, 1280, 3), dtype=np.uint8)
                cv2.putText(img, "NO CAMERA", (100, 360), cv2.FONT_HERSHEY_SIMPLEX, 2, (255,255,255), 4)
                _, jpg = cv2.imencode('.jpg', img)
                with _lock: _latest_jpeg = jpg.tobytes()
                time.sleep(0.1)

threading.Thread(target=CameraManager().run, daemon=True).start()

class OCRProcessor:
    @staticmethod
    def preprocess_image(image_path):
        import cv2
        import numpy as np
        img = cv2.imread(image_path)
        if img is None: return None
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        denoised = cv2.fastNlMeansDenoising(gray, None, 10, 7, 21)
        binary = cv2.adaptiveThreshold(denoised, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, 11, 2)
        return binary

    @staticmethod
    def verify_location_only(processed_img):
        import pytesseract
        try:
            text = pytesseract.image_to_string(processed_img, config='--psm 6')
        except Exception: return False, "OCR Failed"
        clean_text = re.sub(r'[^A-Z0-9\s]', ' ', text.upper())
        city_match = difflib.get_close_matches("VALENZUELA", clean_text.split(), n=1, cutoff=0.75)
        brgy_match = None
        for b in VALID_BARANGAYS:
            if b in clean_text: brgy_match = b; break
        valid = bool(city_match or brgy_match)
        reason = "Location Not Found"
        if brgy_match: reason = f"Verified: {brgy_match}"
        elif city_match: reason = "Verified: Valenzuela City"
        else: reason = "Address Invalid"
        return valid, reason

def process_job(path, jid):
    try:
        proc = OCRProcessor.preprocess_image(str(path))
        valid, reason = OCRProcessor.verify_location_only(proc)
        _ocr_results[jid] = {"status": "done", "valid": valid, "reason": reason}
    except Exception: _ocr_results[jid] = {"status": "error"}

@app.route("/admin/print_report", methods=["POST"])

        img_path = WEB_STATIC.joinpath("logo-karuhatan.png")
        if img_path.exists():

                b64 = base64.b64encode(f.read()).decode('utf-8')
                data_uri = f"data:image/png;base64,{b64}"

            'margin-bottom': '0.5in', 'margin-left': '0.5in',
            'encoding': "UTF-8", 'no-background': None 
        }

        print(f"Log: Printing Admin Report {pdf_path}")
        subprocess.check_output(["lp", "-d", PRINTER_NAME, str(pdf_path)])
        
        return jsonify({"ok": True})
    except Exception as e:

        return jsonify({"ok": False, "error": str(e)})

@app.route("/api/logs")

@app.route("/api/export_csv")

        c = conn.cursor()

        output.headers["Content-type"] = "text/csv"

        return output
    except Exception as e:
        return str(e)

@app.route("/mjpeg")
def mjpeg():
    def gen():
        while True:
            with _lock: frame = _latest_jpeg
            if frame: yield (MJPEG_BOUNDARY + b'\r\nContent-Type: image/jpeg\r\nContent-Length: ' + str(len(frame)).encode() + b'\r\n\r\n' + frame + b'\r\n')
            time.sleep(0.05)
    return Response(stream_with_context(gen()), mimetype='multipart/x-mixed-replace; boundary=frame')

@app.route("/print", methods=["POST"])
def pt():
    try:
        data = request.json
        html_content = data.get('html')
        meta = data.get('meta', {})

        if not html_content: return jsonify({"ok": False, "error": "No HTML"})
        
        if meta:
            log_transaction(
                doc_type=meta.get('docType', 'Unknown'),
                name=meta.get('fullName', 'Unknown'),
                address=meta.get('address', 'Unknown'),
                purpose=meta.get('purpose', 'N/A'),
                amount=meta.get('price', 0)
            )

        img_path = WEB_STATIC.joinpath("logo-karuhatan.png")
        if img_path.exists():
            with open(img_path, "rb") as f:
                b64 = base64.b64encode(f.read()).decode('utf-8')
                data_uri = f"data:image/png;base64,{b64}"
                html_content = html_content.replace('__LOGO_DATA_URI__', data_uri)
        
        pdf_filename = f"print_{int(time.time())}.pdf"
        pdf_path = TEMP_DIR.joinpath(pdf_filename)
        
        options = {
            'page-size': 'Letter',
            'margin-top': '0.75in', 'margin-right': '0.75in',
            'margin-bottom': '0.75in', 'margin-left': '0.75in',
            'encoding': "UTF-8", 'enable-local-file-access': None, 'no-background': None 
        }
        pdfkit.from_string(html_content, str(pdf_path), options=options)
        
        result = subprocess.check_output(["lp", "-d", PRINTER_NAME, str(pdf_path)], stderr=subprocess.STDOUT).decode('utf-8')
        job_id_match = re.search(r"request id is ([\w-]+)", result)
        job_id = job_id_match.group(1) if job_id_match else None
        
        return jsonify({"ok": True, "job_id": job_id})
    except Exception as e:
        return jsonify({"ok": False, "error": str(e)})

@app.route("/print/status")
def print_status():
    job_id = request.args.get('job_id')
    if not job_id: return jsonify({"status": "completed"}) 
    try:
        output = subprocess.check_output(["lpstat", "-o"], stderr=subprocess.STDOUT).decode('utf-8')
        if job_id in output:
            return jsonify({"status": "printing"})
        else:
            return jsonify({"status": "completed"})
    except:
        return jsonify({"status": "completed"})
